Installation & Quickstart

Shell — Install Core or Full Extras
# Core runtime (only numpy, scipy, pandas — zero heavy dependencies)
pip install -e .

# With plotting, rich console tables, Polars, scikit-learn ML learners, and validation tools
pip install -e ".[all,dev]"
Python — Explicit DesignSpec Quickstart
import quasi

# 1. Declare an explicit, immutable statistical design (validated before fitting)
spec = quasi.PanelDesign(
    outcome="score",
    unit="hospital",
    time="year",
    first_treated="first_treated",
    covariates=("case_mix",),
    est_method="dr_improved",
    control_group="not_yet_treated",
    base_period="universal",
)

# 2. Estimate, inspect summary, and serialize audited JSON provenance
result = quasi.estimate(df, design=spec)
print(result.summary())
record = quasi.ResultRecord.from_json(result.to_json())
print("Identification status:", record.identification_status)  # always 'not_established'
